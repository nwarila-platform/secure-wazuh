# =============================================================================
# Development Dependencies — RHEL/Rocky 8 (ansible-core 2.16) toolchain
# =============================================================================
# This repo deploys to STIG RHEL/Rocky 8 targets running platform-python 3.6.8 —
# the only interpreter on RHEL 8 carrying the libselinux/dnf/firewalld C-bindings
# the roles need. The CONTROLLER ansible-core must stay on the last line that can
# still manage a 3.6 target (2.16.x); ansible-core/ansible-lint 25.x require 2.17+
# and emit `from __future__ import annotations`, which SyntaxErrors on 3.6.
# Runtime collection pins (community.general<8, ansible.posix<2) live in
# ansible/requirements.yml.
#
# Update via: pip install --upgrade -r requirements-dev.txt
# =============================================================================

# YAML linting
yamllint==1.35.1

# Ansible linting + syntax checking (24.x supports ansible-core 2.16; 25.x needs 2.17+)
ansible-lint>=24.2,<25
ansible-core>=2.16,<2.17

# Pre-commit hook framework
pre-commit==4.0.1

# boto3/botocore on the CONTROLLER: amazon.aws.sts_assume_role mints artifact-reader sessions,
# the framework's s3_artifact module signs locally and retrieves the dashboard listener pair, the
# dynamic inventory queries EC2, and the delegated EBS Function-tag resolver runs there. Targets
# need no AWS SDK.
boto3
botocore

# pywinrm on the CONTROLLER: the winrm connection plugin imports it, so the WinRM leg of the
# transport matrix (terraform/aws.tfvars, Connection = "winrm-direct") fails at connect without it
# — "winrm or requests is not installed" — while every SSH leg is unaffected. ntlm transport pulls
# requests-ntlm through this package's own extras. Targets need nothing; WS-Management is in-box.
pywinrm>=0.5.0
